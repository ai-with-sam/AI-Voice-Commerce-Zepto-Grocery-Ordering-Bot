# ================= STEP 2: VOICE HANDLER =================
import asyncio
from functools import partial

def transcribe_audio(file_path):
    result = model.transcribe(file_path)
    return result

async def handle_voice(update: Update, context: ContextTypes.DEFAULT_TYPE):
    user_id = update.message.chat_id

    voice = update.message.voice
    file = await context.bot.get_file(voice.file_id)

    ogg_file = f"{user_id}.ogg"
    wav_file = f"{user_id}.wav"

    await file.download_to_drive(ogg_file)

    # Convert to wav (non-blocking)
    loop = asyncio.get_running_loop()

    await loop.run_in_executor(
        None,
        subprocess.run,
        ["ffmpeg", "-y", "-i", ogg_file, wav_file]
    )

    # Whisper transcription (non-blocking)
    loop = asyncio.get_running_loop()

    result = await loop.run_in_executor(
        None,
        partial(transcribe_audio, wav_file)
    )

    text = result["text"]

    print("Voice Text:", text)

    # Extract item
    item = extract_item(text)

    user_data_store[user_id] = {
        "item": item
    }

    await update.message.reply_text(
        f"🛒 Item detected: {item}\n\nHow many quantity?"
    )
    try:
        os.remove(ogg_file)
        os.remove(wav_file)
    except:
        pass

# ================= ITEM EXTRACTION =================
def extract_item(text):
    text = text.lower()

    # Normalize common variations → Apple Shima
    if any(word in text for word in ["apple", "shimla", "simla", "shima", "simula"]):
        return "Apple Shimla"

    # Remove filler words
    text = re.sub(r"(order|buy|get|me|please)", "", text)

    return text.strip().title()

# ================= STEP 3: QUANTITY =================
async def handle_text(update: Update, context: ContextTypes.DEFAULT_TYPE):
    user_id = update.message.chat_id
    text = update.message.text.strip()

    # Step 3: Quantity handling
    if user_id in user_data_store and "quantity" not in user_data_store[user_id]:
        quantity = extract_quantity(text)

        user_data_store[user_id]["quantity"] = quantity
        item = user_data_store[user_id]["item"]

        await update.message.reply_text(
            f"✅ Item: {item}\n🔢 Quantity: {quantity}\n\nProcessing order..."
        )

        total, screenshot = await asyncio.to_thread(run_zepto, item, quantity, user_id)

        user_data_store[user_id]["total"] = total
        user_data_store[user_id]["screenshot"] = screenshot

        await update.message.reply_text(
            f"💰 Total Amount: {total}\n\nReply YES to continue payment"
        )

    # ✅ STEP 6: Handle YES
    elif text.lower() in ["yes", "y", "ok", "okay"]:
        data = user_data_store.get(user_id)
        instance = playwright_instances.get(user_id)

        if data and instance:
            await context.bot.send_photo(
                chat_id=user_id,
                photo=open(data["screenshot"], "rb"),
                caption="📸 Scan QR and pay"
            )

            await update.message.reply_text("Waiting for payment confirmation...")

            page = instance["page"]

            # Wait for payment in background
            await asyncio.to_thread(
                page.wait_for_selector,
                "text=Order placed",
                timeout=300000
            )

            await update.message.reply_text("✅ Payment Successful!")

            # Cleanup
            instance["browser"].close()
            instance["playwright"].stop()
            playwright_instances.pop(user_id, None)
            user_data_store.pop(user_id, None)

        else:
            await update.message.reply_text(
                "⚠️ No active order found. Please start again."
            )

    else:
        await update.message.reply_text(
            "⚠️ Please send a voice note to start ordering."
        )
# ================= QUANTITY EXTRACTION =================
def extract_quantity(text):
    match = re.search(r"\d+", text)
    return int(match.group()) if match else 1

import asyncio

from telegram.ext import ApplicationBuilder, CommandHandler, MessageHandler, filters

def main():

    from pytz import utc   # 👈 add this

    asyncio.set_event_loop(asyncio.new_event_loop())

    app = ApplicationBuilder().token(TELEGRAM_TOKEN).build()

    # 👇 FIX: force pytz timezone
    app.job_queue.scheduler.configure(timezone=utc)

    app.add_handler(CommandHandler("start", start))
    app.add_handler(MessageHandler(filters.VOICE, handle_voice))
    app.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, handle_text))

    app.run_polling()

if __name__ == "__main__":
    main()
